cmake_minimum_required(VERSION 3.20)
project(exec-plus-plus VERSION 0.2.1 LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "exec++ targets Windows. Use a Windows compiler or a Windows cross-toolchain.")
endif()

add_library(execpp SHARED src/Plugin.cpp src/OutputPanel.cpp src/Process.cpp src/CommandText.cpp)
target_compile_features(execpp PRIVATE cxx_std_17)
target_compile_definitions(execpp PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_include_directories(execpp PRIVATE vendor/notepad-plus-plus vendor/scintilla)
target_link_libraries(execpp PRIVATE comctl32 user32 gdi32)
set_target_properties(execpp PROPERTIES OUTPUT_NAME "exec++" PREFIX "")

if(MSVC)
    set_property(TARGET execpp PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(execpp PRIVATE /W4 /utf-8)
else()
    target_compile_options(execpp PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(execpp PRIVATE -static)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(process-smoke tests/ProcessSmoke.cpp src/Process.cpp src/CommandText.cpp)
    target_include_directories(process-smoke PRIVATE src)
    target_compile_features(process-smoke PRIVATE cxx_std_17)
    target_compile_definitions(process-smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    if(MSVC)
        set_property(TARGET process-smoke PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_options(process-smoke PRIVATE /W4 /utf-8)
    else()
        target_link_options(process-smoke PRIVATE -static)
    endif()
    add_test(NAME process-smoke COMMAND process-smoke)
    set_tests_properties(process-smoke PROPERTIES TIMEOUT 90)
    find_program(PWSH_EXECUTABLE pwsh)
    if(PWSH_EXECUTABLE)
        add_test(NAME powershell7-smoke COMMAND process-smoke --pwsh)
        set_tests_properties(powershell7-smoke PROPERTIES TIMEOUT 30)
    endif()
endif()
